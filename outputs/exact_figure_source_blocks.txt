    8: def wilson_ci(k, n, z=1.959963984540054):
    9:     if n == 0:
   10:         return (0.0, 0.0)
   11:     p = k / n
   12:     denom = 1.0 + (z**2) / n
   13:     center = (p + (z**2) / (2*n)) / denom
   14:     margin = (z / denom) * math.sqrt((p*(1-p)/n) + (z**2)/(4*n**2))
   15:     lower = max(0.0, center - margin)
   16:     upper = min(1.0, center + margin)
   17:     return lower, upper
